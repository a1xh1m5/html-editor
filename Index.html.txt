  <script>
    const starterCode = `<!DOCTYPE html>
<html>
<head>
  <style>
    body {
      font-family: sans-serif;
      padding: 20px;
      background: #f4f4f9;
      color: #333;
    }
    h1 { color: #4a90e2; }
    .card {
      background: white;
      padding: 16px;
      border-radius: 8px;
      box-shadow: 0 4px 6px rgba(0,0,0,0.1);
      margin-bottom: 20px;
    }
    iframe {
      width: 100%;
      height: 250px;
      border: 1px solid #ccc;
      border-radius: 6px;
    }
  </style>
</head>
<body>
  <h1>Live Studio</h1>
  <div class="card">
    <p>Edit the code on the right to see updates here instantly.</p>
  </div>

  <h2>Embedded iFrame Test</h2>
  <iframe src="https://example.com" title="Example iFrame"></iframe>
</body>
</html>`;

    document.getElementById('code-input').value = starterCode;

    // Initialize CodeMirror editor
    const editor = CodeMirror.fromTextArea(document.getElementById('code-input'), {
      mode: 'htmlmixed',
      theme: 'dracula',
      lineNumbers: true,
      autoCloseTags: true,
      autoCloseBrackets: true,
      lineWrapping: true,
      tabSize: 2
    });

    const iframe = document.getElementById('preview-frame');

    // Function to render preview with iframe capability
    function updatePreview() {
      const code = editor.getValue();
      const iframeDoc = iframe.contentDocument || iframe.contentWindow.document;
      iframeDoc.open();
      iframeDoc.write(code);
      iframeDoc.close();
    }

    // Debounce preview updates to maintain UI performance
    let timeout;
    editor.on('change', () => {
      clearTimeout(timeout);
      timeout = setTimeout(updatePreview, 250);
    });

    // Initial render
    updatePreview();

    // Open render output in a brand new browser tab via Blob URL
    function openInNewTab() {
      const code = editor.getValue();
      const blob = new Blob([code], { type: 'text/html' });
      const url = URL.createObjectURL(blob);
      window.open(url, '_blank');
    }
  </script>
